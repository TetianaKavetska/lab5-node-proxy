# Перевірка GET: перший запит
curl -s -D - -o /dev/null http://127.0.0.1:3000/418

# Перевірка GET: повторний запит
curl -s -D - -o /dev/null http://127.0.0.1:3000/418

# Перевірка непідтримуваного методу HEAD
curl -I http://127.0.0.1:3000/200

# Перевірка DELETE
curl -X DELETE http://127.0.0.1:3000/201 -i

# Повторний DELETE неіснуючої картинки
curl -X DELETE http://127.0.0.1:3000/201 -i

# Перевірка неправильного шляху
curl -i http://127.0.0.1:3000/abc

# Перевірка недозволеного методу
curl -i -X POST http://127.0.0.1:3000/200

# Перевірка відсутньої картинки
curl -i http://127.0.0.1:3000/999

# Перевірка роботи сервера після помилок
curl -s -D - -o /dev/null http://127.0.0.1:3000/200
